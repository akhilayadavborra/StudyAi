// ║                                                                  ║
// ║  STEP 1 — Create a Google Cloud Project                         ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  1. Go to https://console.cloud.google.com                      ║
// ║  2. Click the project dropdown (top-left) → "New Project"       ║
// ║  3. Name it "StudyAI" → click Create                            ║
// ║                                                                  ║
// ║  STEP 2 — Enable the Google Identity API                        ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  1. In your project, go to:                                     ║
// ║     APIs & Services → Library                                   ║
// ║  2. Search "Google Identity" → Enable it                        ║
// ║                                                                  ║
// ║  STEP 3 — Configure OAuth Consent Screen                        ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  1. Go to: APIs & Services → OAuth consent screen               ║
// ║  2. Choose "External" → Create                                  ║
// ║  3. Fill in:                                                    ║
// ║       App name:        StudyAI                                  ║
// ║       User support email: your Gmail                            ║
// ║       Developer contact:  your Gmail                            ║
// ║  4. Click Save & Continue through all steps                     ║
// ║  5. On "Test users" step → Add your Gmail address               ║
// ║     (required while app is in Testing mode)                     ║
// ║                                                                  ║
// ║  STEP 4 — Create OAuth 2.0 Credentials                         ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  1. Go to: APIs & Services → Credentials                        ║
// ║  2. Click "+ Create Credentials" → OAuth client ID              ║
// ║  3. Application type: Web application                           ║
// ║  4. Name: StudyAI Web Client                                    ║
// ║  5. Under "Authorized JavaScript origins" add:                  ║
// ║       http://localhost:3000                                      ║
// ║       http://localhost:5173   (if using Vite)                   ║
// ║  6. Leave "Authorized redirect URIs" empty (not needed for GIS) ║
// ║  7. Click Create → Copy your Client ID                          ║
// ║                                                                  ║
// ║  STEP 5 — Paste Your Client ID Below                            ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  Find the line:                                                  ║
// ║    const GOOGLE_CLIENT_ID = "YOUR_GOOGLE_CLIENT_ID_HERE";       ║
// ║  Replace it with:                                                ║
// ║    const GOOGLE_CLIENT_ID = "1234567890-abc...apps.googleusercontent.com"; ║
// ║                                                                  ║
// ║  STEP 6 — Run Your App Locally                                  ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  npm install && npm start   (Create React App)                  ║
// ║  — or —                                                         ║
// ║  npm install && npm run dev (Vite)                              ║
// ║                                                                  ║
// ║  Then open http://localhost:3000 and click "Continue with       ║
// ║  Google" — it will show all Google accounts signed into         ║
// ║  your browser/device via the official Google chooser.           ║
// ║                                                                  ║
// ║  ⚠️  WHY IT WON'T WORK IN THE CLAUDE PREVIEW                   ║
// ║  ─────────────────────────────────────────────────────────────  ║
// ║  Claude's artifact sandbox runs inside an iframe with strict    ║
// ║  security policies. Google's OAuth popup is blocked by these    ║
// ║  iframe restrictions — this is expected. The code is correct;   ║
// ║  it only works when served from localhost or your own domain.   ║
// ║                                                                  ║
// ║  HOW ACCOUNTS ARE SHOWN                                         ║
// ║  ─────────────────────────────────────────────────────────────  ║

// DATA STORE — starts completely EMPTY

const useStore = () => {
  const [user, setUser] = useState(null); // null until logged in
  const [assignments, setAssignments] = useState([]);
  const [attendance, setAttendance] = useState([]);
  const [studyTasks, setStudyTasks] = useState([]);
  const [streak, setStreak] = useState(0);
  const [streakUpdating, setStreakUpdating] = useState(false);

  const login = (userData) => setUser({ ...userData, streak: userData.streak ?? 0 });

  // Increment streak with animation flag
  const incrementStreak = useCallback(() => {
    setStreak(s => s + 1);
    setStreakUpdating(true);
    setTimeout(() => setStreakUpdating(false), 600);
    // Sync back to user object
    setUser(u => u ? { ...u, streak: (u.streak ?? 0) + 1 } : u);
  }, []);

  const resetStreak = useCallback(() => {
    setStreak(0);
    setUser(u => u ? { ...u, streak: 0 } : u);
  }, []);

  const addAssignment = (a) => setAssignments(prev => [...prev, { ...a, id: Date.now(), status: "pending" }]);
  const deleteAssignment = (id) => setAssignments(prev => prev.filter(a => a.id !== id));
  const updateAssignment = (id, updates) => setAssignments(prev => prev.map(a => a.id === id ? { ...a, ...updates } : a));
  const toggleAssignment = (id) => setAssignments(prev => prev.map(a => a.id === id
    ? { ...a, status: a.status === "completed" ? "pending" : "completed" } : a));

  const addAttendance = (s) => setAttendance(prev => [...prev, { ...s, id: Date.now() }]);
  const updateAttendance = (id, field, val) => setAttendance(prev => prev.map(s => s.id === id ? { ...s, [field]: Number(val) } : s));
  const deleteAttendance = (id) => setAttendance(prev => prev.filter(s => s.id !== id));

  const addStudyTask = (t) => setStudyTasks(prev => [...prev, { ...t, id: Date.now(), completed: false }]);
  const deleteStudyTask = (id) => setStudyTasks(prev => prev.filter(t => t.id !== id));
  const toggleStudyTask = (id) => setStudyTasks(prev => prev.map(t => t.id === id ? { ...t, completed: !t.completed } : t));
  const updateStudyTask = (id, updates) => setStudyTasks(prev => prev.map(t => t.id === id ? { ...t, ...updates } : t));

  const avgAttendance = attendance.length
    ? Math.round(attendance.reduce((a, s) => a + (s.attended / s.total * 100), 0) / attendance.length)
    : 0;

  const stats = {
    attendance: avgAttendance,
    pending: assignments.filter(a => a.status === "pending").length,
    studyHours: studyTasks.filter(t => t.completed).length,
    total: assignments.length,
  };

  return {
    user, login,
    assignments, addAssignment, deleteAssignment, updateAssignment, toggleAssignment,
    attendance, addAttendance, updateAttendance, deleteAttendance,
    studyTasks, addStudyTask, deleteStudyTask, toggleStudyTask, updateStudyTask,
    stats,
    streak, streakUpdating, incrementStreak, resetStreak,
  };
};

// REAL GOOGLE OAUTH — uses Google Identity Services (GIS)
// Loads the official Google GSI script and calls the real
// account chooser. No hardcoded emails. No fake accounts.
// ============================================================
const GOOGLE_CLIENT_ID = "YOUR_GOOGLE_CLIENT_ID_HERE";
// ↑ Replace with your actual Client ID from:
//   https://console.cloud.google.com → APIs & Services → Credentials → OAuth 2.0 Client ID
//   Authorized JavaScript Origins: http://localhost:3000 (or your domain)

const useGoogleAuth = (onSuccess) => {
  const initialized = useRef(false);

  const loadAndInit = useCallback(() => {
    const init = () => {
      if (initialized.current) return;
      initialized.current = true;
      window.google.accounts.id.initialize({
        client_id: GOOGLE_CLIENT_ID,
        callback: (response) => {
          // response.credential is the ID token (JWT)
          // In production: send to your backend to verify with Google's tokeninfo endpoint
          // For demo: decode the JWT payload to get user info
          try {
            const payload = JSON.parse(atob(response.credential.split(".")[1]));
            onSuccess({
              name: payload.name,
              email: payload.email,
              avatar: (payload.name || payload.email).slice(0, 2).toUpperCase(),
              googleLinked: true,
              picture: payload.picture,
            });
          } catch {
            console.error("Failed to decode Google credential");
          }
        },
        use_fedcm_for_prompt: true,
      });
    };

    if (window.google?.accounts?.id) {
      init();
    } else {
      // Load the GIS script if not already present
      if (!document.getElementById("google-gsi-script")) {
        const script = document.createElement("script");
        script.id = "google-gsi-script";
        script.src = "https://accounts.google.com/gsi/client";
        script.async = true;
        script.defer = true;
        script.onload = init;
        document.head.appendChild(script);
      }
    }
  }, [onSuccess]);

  const signIn = useCallback(() => {
    loadAndInit();
    // Small delay to ensure GIS is initialized
    setTimeout(() => {
      if (window.google?.accounts?.id) {
        // prompt() opens the real Google One Tap / account chooser
        window.google.accounts.id.prompt((notification) => {
          if (notification.isNotDisplayed() || notification.isSkippedMoment()) {
            // One Tap was blocked (e.g. browser setting). Fall back to popup button flow.
            // Render a real Google Sign-In button in a modal overlay instead.
            window.__gsiPopupFallback && window.__gsiPopupFallback();
          }
        });
      }
    }, 300);
  }, [loadAndInit]);

  return { signIn };
};

          messages: [...messages.map(m => ({ role: m.role, content: m.text })), { role: "user", content: msg }]
        })
      });
      const data = await resp.json();
      const reply = data.content?.map(c => c.text).join("") || "I'm having trouble connecting. Please try again!";
      setMessages(prev => [...prev, { role: "assistant", text: reply }]);
    } catch {
      setMessages(prev => [...prev, { role: "assistant", text: "Connection error. Please check your network and try again. 🔌" }]);
    } finally {
      setLoading(false);
    }
  }, [input, messages, loading]);

  return (
    <div style={{ position: "fixed", inset: 0, zIndex: 500, display: "flex", flexDirection: "column", background: "var(--bg)" }}>
      <div style={{ display: "flex", alignItems: "center", gap: 14, padding: "16px 20px", borderBottom: "1px solid var(--border)", background: "rgba(7,7,15,0.9)", backdropFilter: "blur(20px)" }}>
        <button onClick={onClose} style={{ background: "var(--surface)", border: "1px solid var(--border)", borderRadius: 12, width: 40, height: 40, cursor: "pointer", display: "flex", alignItems: "center", justifyContent: "center", color: "var(--text)" }}>
          <Icon name="close" size={20}/>
        </button>
        <div style={{ width: 44, height: 44, borderRadius: 16, background: "linear-gradient(135deg, var(--purple), var(--cyan))", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 22, boxShadow: "0 0 24px var(--purple-glow)" }}>🤖</div>
        <div>
          <p style={{ fontFamily: "var(--font-display)", fontWeight: 700, fontSize: 16 }}>StudyAI Assistant</p>
          <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
            <div className="dot-online"/>
            <span style={{ fontSize: 12, color: "var(--green)" }}>Online • Powered by Claude</span>
          </div>
        </div>
      </div>

      <div style={{ flex: 1, overflowY: "auto", padding: "20px", display: "flex", flexDirection: "column", gap: 16 }}>
        {messages.map((m, i) => (
          <div key={i} style={{ display: "flex", justifyContent: m.role === "user" ? "flex-end" : "flex-start", gap: 10, alignItems: "flex-end" }}>
            {m.role === "assistant" && <div style={{ width: 32, height: 32, borderRadius: 12, background: "linear-gradient(135deg, var(--purple), var(--cyan))", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 16, flexShrink: 0, boxShadow: "0 0 16px var(--purple-glow)" }}>🤖</div>}
            <div className={`chat-bubble ${m.role === "user" ? "chat-bubble-user" : "chat-bubble-ai"}`} style={{ whiteSpace: "pre-wrap" }}>{m.text}</div>
          </div>
        ))}
        {loading && (
          <div style={{ display: "flex", gap: 10, alignItems: "flex-end" }}>
            <div style={{ width: 32, height: 32, borderRadius: 12, background: "linear-gradient(135deg, var(--purple), var(--cyan))", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 16, flexShrink: 0 }}>🤖</div>
            <div className="chat-bubble chat-bubble-ai" style={{ display: "flex", gap: 4, alignItems: "center", padding: "14px 18px" }}>
              {[0, 1, 2].map(j => <div key={j} style={{ width: 6, height: 6, borderRadius: "50%", background: "var(--purple-light)", animation: `typing-dot 1s ${j * 0.2}s ease-in-out infinite` }}/>)}
            </div>
          </div>
        )}
        <div ref={messagesEndRef}/>
      </div>

      {messages.length <= 1 && (
        <div style={{ padding: "0 20px 16px", display: "flex", gap: 8, overflowX: "auto" }}>
          {quickPrompts.map(q => (
            <button key={q} onClick={() => sendMessage(q)}
              style={{ padding: "8px 14px", borderRadius: 99, border: "1px solid rgba(108,92,231,0.3)", background: "rgba(108,92,231,0.1)", color: "var(--purple-light)", fontSize: 12, fontWeight: 500, cursor: "pointer", whiteSpace: "nowrap", fontFamily: "var(--font-body)", transition: "var(--transition)" }}>
              {q}
            </button>
          ))}
        </div>
      )}

      <div style={{ padding: "12px 20px 24px", borderTop: "1px solid var(--border)", background: "rgba(7,7,15,0.9)", backdropFilter: "blur(20px)" }}>
        <div style={{ display: "flex", gap: 12, background: "rgba(255,255,255,0.05)", border: "1px solid var(--border)", borderRadius: 20, padding: "8px 8px 8px 16px", transition: "var(--transition)" }}>
          <input ref={inputRef} value={input} onChange={e => setInput(e.target.value)}
            onKeyDown={e => e.key === "Enter" && !e.shiftKey && sendMessage()}
            placeholder="Ask me anything about studying..."
            style={{ flex: 1, background: "none", border: "none", outline: "none", color: "var(--text)", fontFamily: "var(--font-body)", fontSize: 14 }}/>
          <button onClick={() => sendMessage()} disabled={!input.trim() || loading}
            style={{ width: 44, height: 44, borderRadius: 14, background: input.trim() ? "linear-gradient(135deg, var(--purple), #5a4bd1)" : "var(--surface)", border: "none", cursor: input.trim() ? "pointer" : "not-allowed", display: "flex", alignItems: "center", justifyContent: "center", transition: "var(--transition)", flexShrink: 0, boxShadow: input.trim() ? "0 4px 16px var(--purple-glow)" : "none", color: "white" }}>
            <Icon name="send" size={18} color={input.trim() ? "white" : "var(--text-faint)"}/>
          </button>
        </div>
      </div>
    </div>
  );
};

// FLOATING ACTION BUTTON WITH MINI MENU
// ============================================================
const FloatingActions = ({ onAddAssignment, onAddStudyTask }) => {
  const [open, setOpen] = useState(false);

  const actions = [
    { label: "Add Assignment", icon: "📝", onClick: () => { setOpen(false); onAddAssignment(); } },
    { label: "Add Study Task", icon: "📚", onClick: () => { setOpen(false); onAddStudyTask(); } },
  ];

  return (
    <>
      {open && (
        <div style={{ position: "fixed", inset: 0, zIndex: 48 }} onClick={() => setOpen(false)}/>
      )}
      {open && (
        <div className="fab-menu">
          {actions.map((a, i) => (
            <div key={a.label} className="fab-menu-item"
              onClick={a.onClick}
              style={{ animation: `fab-menu-item 0.2s ${i * 0.05}s ease both` }}>
              <span style={{ fontSize: 18 }}>{a.icon}</span>
              {a.label}
            </div>
          ))}
        </div>
      )}
      <button className={`fab ${open ? "open" : ""}`} onClick={() => setOpen(o => !o)}>
        +
      </button>
    </>
  );
};

// ============================================================
// MAIN APP
// ============================================================
export default function App() {
  const [screen, setScreen] = useState("splash");
  const [page, setPage] = useState("dashboard");
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [aiOpen, setAiOpen] = useState(false);
  const [addAssignmentModal, setAddAssignmentModal] = useState(false);
  const [addStudyTaskModal, setAddStudyTaskModal] = useState(false);
  const [celebration, setCelebration] = useState(false);
  // Track the "all done" state to prevent repeated triggers
  const prevAllDone = useRef(false);
  // Track if celebration already fired for current "all done" session
  const celebrationFired = useRef(false);
  const store = useStore();

  const handleLogin = (userData) => {
    store.login(userData);
    setScreen("app");
  };

  // Watch ALL tasks (assignments + study tasks) for completion → celebration + streak
  useEffect(() => {
    if (screen !== "app") return;

    const allAssignments = store.assignments;
    const allStudyTasks = store.studyTasks;
    const totalItems = allAssignments.length + allStudyTasks.length;

    // Need at least one item of each type to celebrate
    if (totalItems === 0) {
      prevAllDone.current = false;
      celebrationFired.current = false;
      return;
    }

    const assignmentsDone = allAssignments.length === 0 || allAssignments.every(a => a.status === "completed");
    const studyDone = allStudyTasks.length === 0 || allStudyTasks.every(t => t.completed);
    const allDone = assignmentsDone && studyDone && totalItems > 0;

    if (allDone && !prevAllDone.current && !celebrationFired.current) {
      // First time crossing from incomplete → complete
      celebrationFired.current = true;
      setCelebration(true);
      store.incrementStreak();
    }

    if (!allDone) {
      // User unchecked a task → reset trigger so next completion fires again
      celebrationFired.current = false;
    }

    prevAllDone.current = allDone;
  }, [store.assignments, store.studyTasks, screen]);

  const handleAddAssignment = () => setAddAssignmentModal(true);
  const handleAddStudyTask = () => setAddStudyTaskModal(true);

  const pageComponent = () => {
    switch (page) {
      case "dashboard":
        return <DashboardPage
          user={store.user} stats={store.stats}
          assignments={store.assignments} studyTasks={store.studyTasks}
          streak={store.streak} streakUpdating={store.streakUpdating}
          onMenu={() => setSidebarOpen(true)} onAI={() => setAiOpen(true)}
          onAddAssignment={handleAddAssignment} onAddStudyTask={handleAddStudyTask}/>;
      case "assignments":
        return <AssignmentsPage
          assignments={store.assignments}
          onMenu={() => setSidebarOpen(true)} onAI={() => setAiOpen(true)}
          onAdd={handleAddAssignment}
          onDelete={store.deleteAssignment} onToggle={store.toggleAssignment}
          onUpdate={store.updateAssignment}/>;
      case "attendance":
        return <AttendancePage
          attendance={store.attendance}
          onMenu={() => setSidebarOpen(true)} onAI={() => setAiOpen(true)}
          onAdd={store.addAttendance} onUpdate={store.updateAttendance} onDelete={store.deleteAttendance}/>;
      case "planner":
        return <StudyPlannerPage
          studyTasks={store.studyTasks}
          onMenu={() => setSidebarOpen(true)} onAI={() => setAiOpen(true)}
          onAdd={store.addStudyTask} onDelete={store.deleteStudyTask}
          onToggle={store.toggleStudyTask} onUpdate={store.updateStudyTask}/>;
      case "profile":
        return <ProfilePage
          user={store.user} assignments={store.assignments}
          attendance={store.attendance} studyTasks={store.studyTasks}
          onMenu={() => setSidebarOpen(true)}/>;
      default: return null;
    }
  };

  return (
    <>
      <GlobalStyles/>
      <div style={{ height: "100%", background: "var(--bg)", position: "relative", maxWidth: 480, margin: "0 auto", overflow: "hidden", display: "flex", flexDirection: "column" }}>
        <div style={{ position: "absolute", width: 200, height: 200, borderRadius: "50%", background: "radial-gradient(circle, rgba(108,92,231,0.06) 0%, transparent 70%)", top: "30%", left: "50%", transform: "translateX(-50%)", pointerEvents: "none", zIndex: 0 }}/>

        {screen === "splash" && <SplashScreen onDone={() => setScreen("auth")}/>}
        {screen === "auth" && <AuthScreen onLogin={handleLogin}/>}

        {screen === "app" && (
          <div style={{ height: "100%", display: "flex", flexDirection: "column", position: "relative", zIndex: 1 }}>
            <div key={page} className="page-enter" style={{ flex: 1, display: "flex", flexDirection: "column", minHeight: 0 }}>
              {pageComponent()}
            </div>

            {sidebarOpen && (
              <Sidebar page={page} setPage={setPage} user={store.user}
                streak={store.streak} streakUpdating={store.streakUpdating}
                onClose={() => setSidebarOpen(false)}/>
            )}

            {addAssignmentModal && <AddAssignmentModal onClose={() => setAddAssignmentModal(false)} onAdd={store.addAssignment}/>}
            {addStudyTaskModal && <AddStudyTaskModal onClose={() => setAddStudyTaskModal(false)} onAdd={store.addStudyTask}/>}

            {aiOpen ? (
              <AIChat onClose={() => setAiOpen(false)}/>
            ) : (
              <>
                {/* FAB with mini menu — visible on non-dedicated pages */}
                {page !== "assignments" && page !== "planner" && (
                  <FloatingActions onAddAssignment={handleAddAssignment} onAddStudyTask={handleAddStudyTask}/>
                )}
                {/* Floating AI button on inner pages */}
                {(page === "attendance" || page === "profile") && (
                  <button onClick={() => setAiOpen(true)} style={{ position: "fixed", bottom: 100, right: 24, width: 48, height: 48, borderRadius: "50%", background: "linear-gradient(135deg, #1a0050, var(--purple))", border: "1px solid rgba(108,92,231,0.4)", cursor: "pointer", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 20, boxShadow: "0 4px 24px var(--purple-glow)", zIndex: 49 }}>
                    🤖
                  </button>
                )}
              </>
            )}

            {celebration && (
              <CelebrationOverlay
                streak={store.streak}
                onClose={() => setCelebration(false)}
              />
            )}
          </div>
        )}
      </div>
    </>
  );
}
